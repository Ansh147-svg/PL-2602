CI Pipeline;
pipeline {
    agent any
    tools {
        maven 'maven3'
        jdk 'jdk21'
    }


    stages {

        stage('Clone the Code') {
            steps {
                git 'https://github.com/KastroVKiran/Netflix-Pipeline-Project.git'
            }
        }

        stage('Maven Build') {
            steps {
                sh 'mvn clean package -T 1C -DskipTests'
            }
        }

        stage('Archive Artifact') {
            steps {
                echo 'Archiving WAR file in Jenkins...'
                archiveArtifacts artifacts: 'target/*.war', fingerprint: true
            }
        }

        stage('Upload Artifact to S3') {
            steps {
                echo 'Uploading WAR to S3 bucket using EC2 IAM role...'
                sh '''
                    aws s3 cp target/NETFLIX-1.2.3.war \
                    s3://my-unique-bucket-name-12345-kastro/NETFLIX-${BUILD_NUMBER}.war \
                    --region ap-south-1
                '''
            }
        }
    }

    post {
        always {
            echo 'Cleaning up workspace...'
            cleanWs()
        }
    }
}

CD Job - Build 1 ---- I want to pick a specific artifact from the S3 bucket and that artifact i want to deploy

CD Pipeline;
pipeline {
    agent any

    parameters {
        string(
            name: 'BUILD_TO_DEPLOY',
            defaultValue: '1',
            description: 'Build number of artifact in S3 to deploy'
        )
    }

    stages {

        stage('Download Artifact from S3') {
            steps {
                echo "Downloading NETFLIX-${params.BUILD_TO_DEPLOY}.war from S3..."

                sh """
                    aws s3 cp \
                    s3://my-unique-bucket-name-12345-kastro/NETFLIX-${params.BUILD_TO_DEPLOY}.war \
                    ./deploy.war \
                    --region ap-south-1
                """
            }
        }

        stage('Deploy to Tomcat') {
            steps {
                script {
                    echo 'Deploying to Tomcat...'
                    sh 'ls -lh deploy.war'
                    sh 'sudo cp deploy.war /home/ec2-user/apache-tomcat-9.0.113/webapps/NETFLIX.war'
                }
            }
        }
    }

    post {
        always {
            echo 'Cleaning workspace...'
            cleanWs()
        }
    }
}
