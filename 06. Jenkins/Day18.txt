#!/bin/bash
TIMESTAMP=$(date +%F-%H-%M)
BACKUP_DIR="/var/lib/jenkins"
S3_BUCKET="kastro-demo-bucket"
tar -czf /tmp/jenkins-backup-$TIMESTAMP.tar.gz $BACKUP_DIR
aws s3 cp /tmp/jenkins-backup-$TIMESTAMP.tar.gz s3://$S3_BUCKET/
rm -f /tmp/jenkins-backup-$TIMESTAMP.tar.gz

Lets extract the tar file ----> sudo tar -xzf <TarFileName> -C /
-C stands for:
Change directory
It tells tar:
👉 Switch to this directory BEFORE performing the operation

So, tar -xzf backup.tar.gz -C /
Means:
✔ Change to / (root directory)
✔ Then extract files
So files restore into their original absolute paths.
✅ Why -C Exists
Without -C, tar works in current directory.
With -C, you control where extraction happens.

ls ----> You will see the new Jenkins directory
cd jenkins ----> ll ----> cd workspace/ ----> ll ----> You can see the pipeline name which was deleted
sudo systemctl start jenkins
